%% Example CV - Alex Morgan
%% Comprehensive overview of all competencies, experience, and achievements.
%% Use as a master reference when tailoring role-specific CVs.
%%
%% SETUP: Run /setup to populate this with your actual information.
%% Compile with: lualatex main_example.tex
%% (pdflatex often fails on modern MiKTeX with fontawesome5 font-expansion errors.)
%%
%% NOTE: This copy targets a single page by explicit request (easier for HR
%% to skim) - it deliberately overrides the framework's default 2-page CV
%% standard documented in 05-cv-templates.md. If a future tailored CV should
%% go back to 2 pages, drop the extra spacing/font compression below and
%% restore the Core Competencies section.

\documentclass[11pt,a4paper,sans]{moderncv}
\moderncvstyle{banking}
\moderncvcolor{blue}

% banking's default icon set (FontAwesome6 + academicons) renders contact icons as
% broken fallback text (e.g. "MOBILE-ANDROID-ALT") on this system's hand-installed
% FA6 build. marvosym ships in every standard TeX Live install and its phone/email/
% bullet glyphs render reliably, so force it here instead of debugging FA6 glyph
% mapping skew.
\moderncvicons{marvosym}

% Force the name and section headings to render in moderncv blue (color1).
% Default banking leaves them black: moderncvstylebanking.sty's \colorlet
% copies (not aliases) the pre-scheme accent colour, so the name colours are
% frozen before \moderncvcolor runs. Re-let them after. \namefont is the hook
% every name-style macro routes through, so this also works on moderncv 2.3.1
% (Debian/Ubuntu apt), which has no \firstnamestyle/\lastnamestyle at all.
% Sized down from the stock 34/36pt for the single-page header budget below.
\renewcommand*{\namefont}{\fontsize{26}{28}\bfseries\upshape}
\colorlet{firstnamecolor}{color1}
\colorlet{lastnamecolor}{color1}
\colorlet{namecolor}{color1}
\renewcommand*{\sectionstyle}[1]{{\sectionfont\color{color1}#1}}

% Every technology / programming language / architecture-pattern name in body
% text is wrapped in \tech{} for consistent bold+accent-colour emphasis
% throughout the document (profile statement, experience bullets, project
% description) - this is now where the skill list itself lives, since there
% is no separate Core Competencies section on the single-page layout.
\newcommand{\tech}[1]{\textbf{\textcolor{color1}{#1}}}

% microtype's character protrusion/expansion adjusts glyph widths/positions
% at font/color boundaries and at line-stretch points, which was producing
% phantom spaces in pypdf's text-layer extraction ("Tokio" -> "T okio",
% invisible on the rendered page). Disabling both fixed most occurrences.
% A handful of pypdf-only cases remain (confirmed: poppler's pdftotext -
% the project's documented fallback extractor - reads the same PDF cleanly
% with no splits at all), so this is a pypdf parsing quirk rather than an
% actual defect in the PDF; not worth chasing further with more microtype
% settings (kerning=false was also tried and made no difference).
\microtypesetup{protrusion=false,expansion=false}

% A light tint of the accent colour (not a full-saturation block) sits behind
% the header so name/address/icon colours stay legible on it unchanged, and
% so it doesn't fight with the dark-neutral link colour defined below.
\colorlet{headerbandcolor}{color1!12}

% Links (LinkedIn/GitHub/project links) read as the same blue as the name and
% section headings by default - switch them to a dark neutral instead so the
% accent colour stays reserved for headings/\tech{} emphasis.
\definecolor{linkgray}{HTML}{333333}

\usepackage[utf8]{inputenc}
% pdflatex fallback only (the documented engine is lualatex, which skips this
% branch). Without T1 font encoding pdflatex builds accented letters with
% \accent, and the PDF text layer stores them decomposed - `e` + U+0300 rather
% than U+00E8 - so an ATS keyword match on "Genève" fails while the page looks
% right. moderncv 2.5 loads T1 itself under pdflatex; 2.3.1 (Debian/Ubuntu apt)
% does not. \ifpdftex comes from iftex, which every moderncv version loads.
\ifpdftex\usepackage[T1]{fontenc}\fi
% moderncv loads hyperref itself in an \AtEndPreamble hook, so \hypersetup
% must go in an \AtEndPreamble of our own: on moderncv < 2.4 a top-level
% \usepackage{hyperref} clashes with the class's own
% \RequirePackage[unicode]{hyperref}. From 2.4.0 the class passes its options
% through \PassOptionsToPackage instead, which is what removes that clash.
\AtEndPreamble{\hypersetup{
    colorlinks=true,
    linkcolor=linkgray,
    filecolor=linkgray,
    urlcolor=linkgray,
    pdftitle={Alex Morgan - CV},
    % Keep pdfpagemode=UseNone: this block runs after moderncv's own
    % \AtEndPreamble (moderncv.cls sets pdfpagemode there), so a FullScreen
    % value here would win and open every CV in fullscreen presentation mode.
    pdfpagemode=UseNone,
}}
% Slightly tighter margins than the 2-page stock 0.80 scale - part of the
% single-page budget, not a "squeeze to fit" hack on top of unchanged content
% (the content itself was cut - see Professional Experience).
\usepackage[scale=0.85,top=1.3cm,bottom=1.3cm]{geometry}
\usepackage{import}
% Already an indirect moderncv dependency (moderncviconstikz, skill matrices),
% loaded explicitly here since the header band below relies on it directly.
\usepackage{tikz}

% personal data
\name{Alex}{Morgan}
% If you have no address to list, DELETE this whole line. \address{}{}{} fails
% with "There's no line here to end" on every moderncv version.
\address{Palermo, Italy}{}{}
\phone[mobile]{+1 555-0100}
\email{alex.morgan@example.com}
\extrainfo{\href{https://linkedin.com/in/jaster-victor}{\underline{LinkedIn}}, \href{https://github.com/JasterV}{\underline{GitHub}}}
% Rendered inside the header band by the \makehead override below.
\newcommand{\cvlanguages}{Spanish (native)~\textbullet~Catalan (native)~\textbullet~English (native)~\textbullet~Italian (fluent)}

% Redefine moderncv banking's header (originally in moderncvheadiii.sty) to:
%  1. add breathing room between the name/title line and the address line
%     below it (the vendor macro uses a bare `\\`, i.e. one baselineskip -
%     reads as cramped at this name font size);
%  2. paint a light accent-tinted background band behind it spanning the
%     full physical page width (edge to edge, not just the text column) via
%     a tikz page-absolute overlay, so the page doesn't open on a flat white
%     block and there's no white gutter on either side of the band. This is
%     independent of the normal text margins, so \headerbandheight below is
%     tuned by eye against the actual header content, not derived from it.
%  3. append a languages line under the contact details, so languages sit
%     with the rest of the identity/contact info instead of in their own
%     section further down the page.
% \@firstname, \if@details, etc. are internal moderncv commands whose `@` is
% only letter-catcode inside a .sty/.cls - in a .tex document it is "other"
% catcode by default, so without \makeatletter/\makeatother here \@firstname
% parses as the primitive \@ followed by literal text "firstname" (this is
% exactly what broke the header on the first attempt: the name/contact block
% printed as the literal words "firstname lastname | title" with nothing else,
% because the whole \if@details{...} conditional mis-tokenized).
\newlength{\headerbandheight}
\setlength{\headerbandheight}{3.6cm}
\makeatletter
\renewcommand*{\makehead}{%
  \begin{tikzpicture}[overlay]
    \fill[headerbandcolor] (current page.north west) rectangle ([yshift=-\headerbandheight]current page.north east);
  \end{tikzpicture}%
  \setlength{\makeheaddetailswidth}{0.8\textwidth}%
  \hfil%
  \parbox{\makeheaddetailswidth}{%
    \centering%
    \firstnamestyle{\@firstname\ }\lastnamestyle{\@lastname}%
    \ifthenelse{\equal{\@title}{}}{}{\titlestyle{~|~\@title}}%
    \if@details{%
      \\[6pt]%
      \addressfont\color{addresscolor}%
      \ifthenelse{\isundefined{\@addressstreet}}{}{\addtomakeheaddetails{\addresssymbol\@addressstreet}%
        \ifthenelse{\equal{\@addresscity}{}}{}{\addtomakeheaddetails[~--~]{\@addresscity}}%
        \ifthenelse{\equal{\@addresscountry}{}}{}{\addtomakeheaddetails[~--~]{\@addresscountry}}%
        \flushmakeheaddetails\@firstmakeheaddetailselementtrue\\}%
      \ifthenelse{\isundefined{\@born}}{}{\addtomakeheaddetails{\bornsymbol\@born}}%
      \collectionloop{phones}{%
        \addtomakeheaddetails{\csname\collectionloopkey phonesymbol\endcsname\collectionloopitem}}%
      \ifthenelse{\isundefined{\@email}}{}{\addtomakeheaddetails{\emailsymbol\emaillink{\@email}}}%
      \ifthenelse{\isundefined{\@homepage}}{}{\addtomakeheaddetails{\homepagesymbol%
        \ifthenelse{\equal{\@homepageprotocol}{http}}{\httplink{\@homepage}}{\httpslink{\@homepage}}}}%
      \collectionloop{socials}{%
        \addtomakeheaddetails{\csname\collectionloopkey socialsymbol\endcsname\collectionloopitem}}%
      \ifthenelse{\isundefined{\@extrainfo}}{}{\addtomakeheaddetails{\@extrainfo}}%
      \flushmakeheaddetails%
      \\[4pt]%
      {\small\cvlanguages}%
      }\fi%
  }}
\makeatother

\begin{document}
